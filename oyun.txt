    <script>
        /**
         * SES YÖNETİCİSİ
         */
        const audioManager = {
            ctx: null,
            enabled: true,
            init() {
                if (!this.ctx) {
                    this.ctx = new (window.AudioContext || window.webkitAudioContext)();
                }
            },
            playPop() {
                if (!this.enabled) return;
                this.init();
                if (this.ctx.state === 'suspended') this.ctx.resume();
                const osc = this.ctx.createOscillator();
                const gain = this.ctx.createGain();
                osc.type = 'sine';
                osc.frequency.setValueAtTime(800, this.ctx.currentTime);
                osc.frequency.exponentialRampToValueAtTime(100, this.ctx.currentTime + 0.1);
                gain.gain.setValueAtTime(0.1, this.ctx.currentTime);
                gain.gain.exponentialRampToValueAtTime(0.01, this.ctx.currentTime + 0.1);
                osc.connect(gain);
                gain.connect(this.ctx.destination);
                osc.start();
                osc.stop(this.ctx.currentTime + 0.1);
            },
            playDing() {
                if (!this.enabled) return;
                this.init();
                const osc = this.ctx.createOscillator();
                const gain = this.ctx.createGain();
                osc.type = 'triangle';
                osc.frequency.setValueAtTime(1200, this.ctx.currentTime);
                gain.gain.setValueAtTime(0.1, this.ctx.currentTime);
                gain.gain.exponentialRampToValueAtTime(0.001, this.ctx.currentTime + 0.5);
                osc.connect(gain);
                gain.connect(this.ctx.destination);
                osc.start();
                osc.stop(this.ctx.currentTime + 0.5);
            },
            toggle() {
                this.enabled = !this.enabled;
                document.getElementById('mute-btn').innerText = this.enabled ? "SES: AÇIK" : "SES: KAPALI";
            }
        };

        /**
         * OYUN MANTIĞI
         */
        const game = {
            data: {
                points: 0,
                clickCount: 0,
                clickPower: 1,
                autoPower: 0,
                prestigeMultiplier: 1,
                stage: 0,
                boostActive: false,
                boostEndTime: 0,
                upgrades: {
                    auto: { count: 0, baseCost: 15 },
                    click: { count: 0, baseCost: 50 },
                    boost: { count: 0, baseCost: 500 }
                }
            },
            
            // --- RESİM DOSYA İSİMLERİ BURADA ---
            // 5 aşama için elindeki 3 resmi paylaştırdım:
            stages: [
                { name: "TOHUM", limit: 50,   img: "level1.png" }, // Başlangıç
                { name: "FİLİZ", limit: 200,  img: "level1.png" }, // Biraz büyüdü ama hala level1
                { name: "BİTKİ", limit: 1000, img: "level2.png" }, // Orta seviye
                { name: "ÇİÇEK", limit: 5000, img: "level2.png" }, // Çiçek açtı
                { name: "AĞAÇ",  limit: Infinity, img: "level3.png" } // Son seviye
            ],

            lastTime: Date.now(),

            init() {
                this.load();
                this.updateUI();
                this.renderPlant();

                const clickArea = document.getElementById('click-area');
                clickArea.addEventListener('mousedown', (e) => this.handleClick(e));
                clickArea.addEventListener('touchstart', (e) => {
                    e.preventDefault();
                    this.handleClick(e.touches[0]);
                }, {passive: false});

                requestAnimationFrame(() => this.loop());
            },

            handleClick(e) {
                let multiplier = this.data.prestigeMultiplier;
                if (Date.now() < this.data.boostEndTime) multiplier *= 2;

                let isBonus = Math.random() < 0.05;
                let gain = this.data.clickPower * multiplier;
                if (isBonus) gain *= 5;

                this.data.points += gain;
                this.data.clickCount++;

                audioManager.playPop();
                this.createFloatingText(e, `+${Math.floor(gain)}`, isBonus);
                
                if (isBonus) this.showBonusMessage();
                
                this.checkGrowth();
                this.updateUI();
            },

            buyUpgrade(type) {
                const upgrade = this.data.upgrades[type];
                let cost = Math.floor(upgrade.baseCost * Math.pow(1.5, upgrade.count));
                if(type === 'boost') cost = upgrade.baseCost * (upgrade.count + 1);

                if (this.data.points >= cost) {
                    this.data.points -= cost;
                    upgrade.count++;
                    audioManager.playDing();

                    if (type === 'auto') this.data.autoPower += 0.5;
                    else if (type === 'click') this.data.clickPower += 1;
                    else if (type === 'boost') this.activateBoost();

                    this.updateUI();
                    this.save();
                }
            },

            activateBoost() {
                this.data.boostEndTime = Date.now() + 60000;
                document.body.classList.add('sunlight-active');
                setTimeout(() => {
                    document.body.classList.remove('sunlight-active');
                    this.updateUI();
                }, 60000);
            },

            loop() {
                const now = Date.now();
                const dt = (now - this.lastTime) / 1000;
                this.lastTime = now;

                if (this.data.autoPower > 0) {
                    let multiplier = this.data.prestigeMultiplier;
                    if (now < this.data.boostEndTime) multiplier *= 2;
                    this.data.points += (this.data.autoPower * multiplier) * dt;
                }

                if (this.data.clickCount % 5 === 0 || this.data.autoPower > 0) {
                    this.updateUI();
                }

                this.updateProgressRing();
                if (Math.floor(now / 1000) % 10 === 0) this.save();
                requestAnimationFrame(() => this.loop());
            },

            checkGrowth() {
                const nextLimit = this.stages[this.data.stage].limit;
                if (this.data.stage === this.stages.length - 1) {
                    document.getElementById('btn-prestige').classList.remove('hidden');
                    return;
                }
                if (this.data.points >= nextLimit) {
                    this.data.stage++;
                    this.renderPlant();
                    audioManager.playDing();
                }
            },

            prestige() {
                if (!confirm("Sıfırlayıp bonus kazanmak istiyor musun?")) return;
                this.data.prestigeMultiplier *= 1.5;
                this.data.points = 0;
                this.data.clickPower = 1;
                this.data.autoPower = 0;
                this.data.stage = 0;
                this.data.upgrades.auto.count = 0;
                this.data.upgrades.click.count = 0;
                this.data.upgrades.boost.count = 0;
                this.data.boostEndTime = 0;
                document.getElementById('btn-prestige').classList.add('hidden');
                this.renderPlant();
                this.updateUI();
                this.save();
            },

            renderPlant() {
                // Resmi güncelle
                const imgElement = document.getElementById('plant-img');
                const currentImg = this.stages[this.data.stage].img;
                
                // Sadece değişirse güncelle (titremeyi önler)
                if (!imgElement.src.includes(currentImg)) {
                    imgElement.style.opacity = 0; // Geçiş efekti için önce gizle
                    setTimeout(() => {
                        imgElement.src = currentImg;
                        imgElement.style.opacity = 1; // Sonra göster
                    }, 150);
                }
                
                document.getElementById('level-display').innerText = this.stages[this.data.stage].name;
            },

            createFloatingText(e, text, isBonus) {
                const el = document.createElement('div');
                el.classList.add('floating-text');
                el.innerText = text;
                const x = e.clientX || (window.innerWidth / 2);
                const y = e.clientY || (window.innerHeight / 2);
                el.style.left = (x - 10) + 'px';
                el.style.top = (y - 30) + 'px';
                if (isBonus) {
                    el.style.color = '#d4af37';
                    el.style.fontSize = '24px';
                }
                document.body.appendChild(el);
                setTimeout(() => el.remove(), 1000);
            },

            showBonusMessage() {
                const el = document.getElementById('message-area');
                el.style.opacity = 1;
                setTimeout(() => el.style.opacity = 0, 1500);
            },

            updateProgressRing() {
                const circle = document.getElementById('progress-ring');
                const radius = circle.r.baseVal.value;
                const circumference = radius * 2 * Math.PI;
                let currentLimit = this.stages[this.data.stage].limit;
                if (currentLimit === Infinity) {
                    circle.style.strokeDashoffset = 0;
                    return;
                }
                let prevLimit = this.data.stage > 0 ? this.stages[this.data.stage - 1].limit : 0;
                let range = currentLimit - prevLimit;
                let currentInStage = this.data.points - prevLimit;
                let percent = Math.min(100, Math.max(0, (currentInStage / range) * 100));
                const offset = circumference - (percent / 100) * circumference;
                circle.style.strokeDashoffset = offset;
            },

            updateUI() {
                document.getElementById('score-display').innerText = Math.floor(this.data.points).toLocaleString();
                let multiplier = this.data.prestigeMultiplier;
                if (Date.now() < this.data.boostEndTime) multiplier *= 2;
                document.getElementById('sps-display').innerText = (this.data.autoPower * multiplier).toFixed(1);
                document.getElementById('prestige-display').innerText = this.data.prestigeMultiplier.toFixed(1);

                this.updateBtn('auto', 'cost-auto', 'btn-auto');
                this.updateBtn('click', 'cost-click', 'btn-click');
                this.updateBtn('boost', 'cost-boost', 'btn-boost');

                if (Date.now() < this.data.boostEndTime) {
                    const timeLeft = Math.ceil((this.data.boostEndTime - Date.now()) / 1000);
                    document.getElementById('btn-boost').disabled = true;
                    document.getElementById('cost-boost').innerText = `${timeLeft}s`;
                }
            },

            updateBtn(type, costId, btnId) {
                const upgrade = this.data.upgrades[type];
                let cost = Math.floor(upgrade.baseCost * Math.pow(1.5, upgrade.count));
                if(type === 'boost') cost = upgrade.baseCost * (upgrade.count + 1);
                const elCost = document.getElementById(costId);
                const elBtn = document.getElementById(btnId);
                if (type !== 'boost' || Date.now() > this.data.boostEndTime) {
                    elCost.innerText = cost + " SP";
                    elBtn.disabled = this.data.points < cost;
                }
            },

            save() { localStorage.setItem('tapSeedSave', JSON.stringify(this.data)); },
            load() {
                const saved = localStorage.getItem('tapSeedSave');
                if (saved) this.data = { ...this.data, ...JSON.parse(saved) };
            },
            resetSave() {
                if(confirm("Sıfırlamak istiyor musun?")) {
                    localStorage.removeItem('tapSeedSave');
                    location.reload();
                }
            }
        };

        window.onload = () => game.init();
    </script>
